```python
# scripts/prepare_ixi.py
"""Download IXI T1+T2, skull-strip, resample to 240x240x155 1mm, split train/val."""
import os, urllib.request, glob, subprocess, argparse
import nibabel as nib
import numpy as np
from scipy.ndimage import zoom

IXI_T1_URL = "https://biomedic.doc.ic.ac.uk/brain-development/downloads/IXI/IXI-T1.tar"
IXI_T2_URL = "https://biomedic.doc.ic.ac.uk/brain-development/downloads/IXI/IXI-T2.tar"

TARGET_SHAPE = (240, 240, 155)

def download_and_extract(url, out_dir):
    fname = os.path.basename(url)
    fpath = os.path.join(out_dir, fname)
    if not os.path.exists(fpath):
        print(f'Downloading {fname}...')
        urllib.request.urlretrieve(url, fpath)
    print(f'Extracting {fname}...')
    subprocess.run(['tar', 'xf', fpath, '-C', out_dir], check=True)

def resample_volume(img, target_shape):
    factors = [t / s for t, s in zip(target_shape, img.shape)]
    return zoom(img, factors, order=1)

def normalize(img):
    nonzero = img[img > 0]
    if len(nonzero) > 0:
        return (img - nonzero.mean()) / (nonzero.std() + 1e-8)
    return img

def process_subject(t1_path, t2_path, out_dir, subj_id):
    t1_img = nib.load(t1_path).get_fdata().astype(np.float32)
    t2_img = nib.load(t2_path).get_fdata().astype(np.float32)

    # Resample to target shape
    t1_rs = resample_volume(t1_img, TARGET_SHAPE)
    t2_rs = resample_volume(t2_img, TARGET_SHAPE)

    # Normalize
    t1_rs = normalize(t1_rs)
    t2_rs = normalize(t2_rs)

    # Create 4-channel: [T1, T1_copy(as T1ce), T2, T2_copy(as FLAIR)]
    volume = np.stack([t1_rs, t1_rs, t2_rs, t2_rs], axis=0)  # [4, D, H, W]

    # Save as single .npz for fast loading
    subj_dir = os.path.join(out_dir, subj_id)
    os.makedirs(subj_dir, exist_ok=True)
    np.savez_compressed(os.path.join(subj_dir, 'data.npz'), image=volume)

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--raw-dir', default='/content/IXI_raw')
    parser.add_argument('--out-dir', default='/content/IXI_processed')
    parser.add_argument('--train-ratio', type=float, default=0.85)
    args = parser.parse_args()

    os.makedirs(args.raw_dir, exist_ok=True)
    download_and_extract(IXI_T1_URL, args.raw_dir)
    download_and_extract(IXI_T2_URL, args.raw_dir)

    # Match T1 and T2 by subject ID
    t1_files = sorted(glob.glob(os.path.join(args.raw_dir, 'IXI*-T1.nii.gz')))
    t2_files = {os.path.basename(f).replace('-T2.nii.gz', ''): f
                for f in glob.glob(os.path.join(args.raw_dir, 'IXI*-T2.nii.gz'))}

    pairs = []
    for t1 in t1_files:
        subj = os.path.basename(t1).replace('-T1.nii.gz', '')
        if subj in t2_files:
            pairs.append((subj, t1, t2_files[subj]))
    print(f'Found {len(pairs)} subjects with both T1 and T2')

    # Process
    for i, (subj, t1, t2) in enumerate(pairs):
        print(f'[{i+1}/{len(pairs)}] {subj}')
        try:
            process_subject(t1, t2, args.out_dir, subj)
        except Exception as e:
            print(f'  SKIP: {e}')

    # Split
    processed = sorted([d for d in os.listdir(args.out_dir)
                        if os.path.isdir(os.path.join(args.out_dir, d))])
    n_train = int(len(processed) * args.train_ratio)
    train_ids = processed[:n_train]
    val_ids = processed[n_train:]

    for split, ids in [('train', train_ids), ('val', val_ids)]:
        split_dir = os.path.join(args.out_dir, split)
        os.makedirs(split_dir, exist_ok=True)
        for sid in ids:
            src = os.path.join(args.out_dir, sid)
            dst = os.path.join(split_dir, sid)
            if not os.path.exists(dst):
                os.rename(src, dst)
        print(f'{split}: {len(ids)} subjects')

if __name__ == '__main__':
    main()
```

```python
# data/ixi_dataset.py
"""IXI dataset loader for SSL pretraining. Loads preprocessed 4-channel .npz files."""
import os, glob, torch, numpy as np
from torch.utils.data import Dataset

class IXIDataset(Dataset):
    def __init__(self, data_dir, split='train', transform=None):
        self.transform = transform
        split_dir = os.path.join(data_dir, split)
        self.files = sorted(glob.glob(os.path.join(split_dir, '*', 'data.npz')))

    def __len__(self):
        return len(self.files)

    def __getitem__(self, idx):
        data = np.load(self.files[idx])
        image = torch.from_numpy(data['image'].astype(np.float32))  # [4, D, H, W]
        # Dummy mask and text for compatibility with transforms
        mask = torch.zeros(image.shape[1:], dtype=torch.long)
        if self.transform:
            image, mask = self.transform(image, mask)
        return {'image': image, 'mask': mask,
                'text_ids': torch.zeros(1, dtype=torch.long),
                'attention_mask': torch.zeros(1, dtype=torch.long)}
```
